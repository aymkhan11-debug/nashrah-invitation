        // Audio Toggle

        let isPlaying = false;

        function toggleAudio() {

            const audio = document.getElementById('bgAudio');

            const toggleBtn = document.getElementById('audioToggle');

            if (isPlaying) {

                audio.pause();

                toggleBtn.innerText = "🔇";

                isPlaying = false;

            } else {

                audio.play();

                toggleBtn.innerText = "🎵";

                isPlaying = true;

            }
